% abmi_header.tex
% LaTeX header for export_abmi_document() PDFs (xelatex), included with pandoc -H.
% ABMI look: Montserrat and Lora type and the navy/sage palette of the ABMI
% reference docs, the institute name in the header, and the logo and landscape
% footer art taken from the reference doc. export_abmi_document() replaces
% @@LOGO@@, @@FOOTER@@ and @@LINK@@ before use; an empty image path leaves that
% image out.

% Fonts ----------------------------------------------------------------------
% Montserrat and Roboto Mono ship with TeX Live (loaded by file name). The serif
% (@@SERIF@@) is Lora when export_abmi_document() finds its font files on the
% system, otherwise Libertinus Serif (TeX Live).
\usepackage{fontspec}
\setmainfont{Montserrat}[Extension=.otf, UprightFont=*-Light,
  ItalicFont=*-LightItalic, BoldFont=*-SemiBold, BoldItalicFont=*-SemiBoldItalic]
\newfontfamily\abmimedium{Montserrat}[Extension=.otf, UprightFont=*-Medium,
  ItalicFont=*-MediumItalic, BoldFont=*-Bold]
\newfontfamily\abmiregular{Montserrat}[Extension=.otf, UprightFont=*-Regular,
  ItalicFont=*-Italic, BoldFont=*-SemiBold]
@@SERIF@@
\setmonofont{RobotoMono}[Extension=.otf, UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic, Scale=MatchLowercase]

% Colours --------------------------------------------------------------------
\usepackage{xcolor}
\definecolor{abminavy}{HTML}{324458}
\definecolor{abmisage}{HTML}{A3A983}
\definecolor{abmigrey}{HTML}{929990}
\definecolor{abmilink}{HTML}{@@LINK@@}
\AtBeginDocument{\color{abminavy}}

% Headings -------------------------------------------------------------------
% (pandoc is run with -V subparagraph so its own \paragraph tweak doesn't clash.)
\usepackage{titlesec}
\titleformat{\section}
  {\fontsize{20}{24}\selectfont\color{abminavy}}{\thesection}{0.6em}{}
\titleformat{\subsection}
  {\abmiserif\itshape\fontsize{14}{17}\selectfont\color{abmisage}}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection}
  {\abmiregular\fontsize{12}{15}\selectfont\color{abminavy}}{\thesubsubsection}{0.6em}{}
\titleformat{\paragraph}[hang]
  {\abmiregular\normalsize\color{abmigrey}}{\theparagraph}{0.6em}{}
\titlespacing*{\section}{0pt}{1.6em}{0.6em}
\titlespacing*{\subsection}{0pt}{1.3em}{0.4em}
\titlespacing*{\subsubsection}{0pt}{1.1em}{0.3em}
\titlespacing*{\paragraph}{0pt}{1em}{0.2em}

% Title block ------------------------------------------------------------------
\makeatletter
\renewcommand{\@maketitle}{%
  \begin{flushleft}
    {\fontsize{26}{31}\selectfont\color{abminavy}\@title\par}
    \ifx\@author\@empty\else
      \vspace{0.8em}{\abmiregular\large\color{abmigrey}\@author\par}\fi
    \ifx\@date\@empty\else
      \vspace{0.3em}{\abmiregular\color{abmigrey}\@date\par}\fi
  \end{flushleft}
  \vspace{1.2em}}
\makeatother

% Captions -------------------------------------------------------------------
\usepackage{caption}
\DeclareCaptionFont{abmi}{\color{abminavy}}
\captionsetup{font={small,abmi}, labelfont=bf, labelsep=period}

% Code -------------------------------------------------------------------------
% Light grey background, and long lines wrap instead of running off the page.
% Set at \begin{document} so it overrides pandoc's highlighting definitions (whose
% default colour scheme has no background, so its Shaded environment is empty).
\usepackage{fvextra}
\usepackage{framed}
\AtBeginDocument{%
  \definecolor{shadecolor}{HTML}{F3F4F6}%
  \DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,commandchars=\\\{\}}%
  \renewenvironment{Shaded}{\begin{snugshade}}{\end{snugshade}}}

% Header and footer ------------------------------------------------------------
\usepackage{graphicx}
\usepackage{fancyhdr}
\usepackage{eso-pic}
\setlength{\headheight}{30pt}
\newcommand{\abmiheadfoot}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyhead[L]{{\abmimedium\fontsize{8}{10}\selectfont\color{abminavy}Alberta Biodiversity Monitoring Institute}%
    \quad{\abmiserif\itshape\fontsize{8}{10}\selectfont\color{abmisage}It's Our Nature to Know}}%
  \fancyhead[R]{\IfFileExists{@@LOGO@@}{\raisebox{-0.35\height}{\includegraphics[height=0.33in]{@@LOGO@@}}}{}}%
  \fancyfoot[R]{{\abmiregular\small\color{abminavy}\thepage}}}
\pagestyle{fancy}
\abmiheadfoot
\fancypagestyle{plain}{\abmiheadfoot}
\AddToShipoutPictureBG{%
  \IfFileExists{@@FOOTER@@}{\AtPageLowerLeft{\includegraphics[width=\paperwidth]{@@FOOTER@@}}}{}}
